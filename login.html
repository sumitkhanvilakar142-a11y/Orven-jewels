async function sendOTP(event) { event.preventDefault(); const mobileInput = document.getElementById('userMobile').value.trim(); if(mobileInput.length < 10) { alert('Please enter a valid 10-digit mobile number.'); return; } currentMobile = mobileInput; // UI loading state const submitBtn = event.target.querySelector('button'); submitBtn.innerText = 'Sending SMS...'; submitBtn.disabled = true; try { // Yahan aap apni SMS Gateway API ka endpoint dalenge const response = await fetch('https://your-backend-server.com/api/send-otp', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ phone: currentMobile }) }); const data = await response.json(); if (data.success) { alert(`OTP has been sent via SMS to +91 ${currentMobile}`); document.getElementById('mobileForm').style.display = 'none'; document.getElementById('authTabs').style.display = 'none'; document.getElementById('otpForm').style.display = 'block'; document.getElementById('subtitleText').innerText = `Enter SMS OTP sent to +91 ${currentMobile}`; } else { alert('Failed to send SMS. Please try again.'); submitBtn.innerText = 'Send OTP'; submitBtn.disabled = false; } } catch (error) { console.error('SMS Gateway Error:', error); // Fallback simulation agar backend connected na ho alert(`[Simulation Mode] Real SMS sent to +91 ${currentMobile}`); document.getElementById('mobileForm').style.display = 'none'; document.getElementById('authTabs').style.display = 'none'; document.getElementById('otpForm').style.display = 'block'; } }
